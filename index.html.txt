        window.onload = function() {

                if(i.getAttribute('onclick').includes(vistaGuardada)) i.classList.add('active');

        function aplicarRolSeguro() {
            var esAdmin = document.getElementById('rol-toggle').checked;
            document.querySelectorAll('.admin-only-item').forEach(el => el.style.display = '');

            if (esAdmin) {
                document.body.classList.remove('modo-socio');
            } else {
                document.body.classList.add('modo-socio');
                var vistaActual = document.querySelector('.page-view.active');
                if (vistaActual && vistaActual.classList.contains('admin-only-item')) {
                    cambiarSeccion('view-dashboard', document.querySelector('.sidebar-item'));
                }
            }
        }

        function cambiarSeccion(seccionId, elem) {
            document.querySelectorAll('.page-view').forEach(v => v.classList.remove('active'));
            document.querySelectorAll('.sidebar-item').forEach(i => i.classList.remove('active'));
            document.querySelectorAll('.mobile-nav-item').forEach(i => i.classList.remove('active'));
            
            var targetView = document.getElementById(seccionId);
            if (targetView) targetView.classList.add('active');
            
            if(elem) elem.classList.add('active');
            
            localStorage.setItem('vistaActual', seccionId);
            
            var dropdown = document.getElementById('mobile-dropdown');
            if(dropdown && dropdown.classList.contains('show')) dropdown.classList.remove('show');

            ejecutarCargas(seccionId);
        }

        function exportarCSV() {
            var tabla = document.getElementById('tabla-exportable');
            var filas = tabla.querySelectorAll('tr');
            var csv = [];
            for (var i = 0; i < filas.length; i++) {
                var celdas = filas[i].querySelectorAll('th, td');
                var filaCsv = [];
                for (var j = 0; j < celdas.length - 1; j++) { 
                    var texto = celdas[j].innerText.replace(/,/g, ''); 
                    filaCsv.push(texto);
                }
                csv.push(filaCsv.join(","));

            var csvFile = new Blob(["\uFEFF" + csv.join("\n")], { type: "text/csv;charset=utf-8;" });

            link.download = "Balance_Directorio_CrediAhorro.csv";
            link.click();

        }

        function emitirCreditoDefinitivo() {
            var s = document.getElementById('credito-socio').value;
            var selSocioTxt = document.getElementById('credito-socio').options[document.getElementById('credito-socio').selectedIndex].text;
            var m = document.getElementById('credito-monto').value;
            var i = document.getElementById('credito-interes').value;
            var c = document.getElementById('credito-cuotas').value;
            var f = document.getElementById('credito-frecuencia').value;
            var fe = document.getElementById('credito-fecha-inicio').value;
            var ho = document.getElementById('credito-hora-inicio').value;
            
            if(!m || !fe) return mostrarToast("Rellene los campos obligatorios.");
            var montoNumericoSolicitado = parseFloat(m) || 0;
            if (montoNumericoSolicitado > liquidezGlobalCalculada) {
                return mostrarToast("❌ ERROR: Liquidez insuficiente en caja. No se puede otorgar este crédito.");
            }
            
            google.script.run.withSuccessHandler(function(r) {
                if(r.status === "OK") {
                    mostrarToast(r.mensaje);
                    document.getElementById('rec-id-cre').innerText = r.idCredito;
                    document.getElementById('rec-socio').innerText = selSocioTxt;
                    document.getElementById('rec-monto').innerText = '$' + parseFloat(m).toFixed(2);
                    document.getElementById('rec-interes').innerText = i + '%';
                    document.getElementById('rec-plazo').innerText = c + ' Cuotas (' + f + ')';
                    document.getElementById('rec-fecha').innerText = fe + ' / ' + ho;
                    document.getElementById('rec-total').innerText = '$' + r.total;
                    document.getElementById('recibo-descargable').style.display = 'block';
                    document.getElementById('recibo-descargable').scrollIntoView({ behavior: 'smooth' });
                    actualizarDashboard();
                    actualizarFondoLiquidoEnCreditos();
                }
            }).otorgarCredito(s, m, i, c, f, fe, ho);
        }

        function cargarSeccionAbonos() {
            var select = document.getElementById('abono-credito');
            select.innerHTML = '<option>Cargando créditos vigentes...</option>';
            google.script.run.withSuccessHandler(function(lista) {
                select.innerHTML = '';
                if(lista.length === 0) {
                    select.innerHTML = '<option value="">No hay créditos vigentes por cobrar</option>'; return;
                }
                lista.forEach(c => { 
                    var opt = document.createElement('option'); opt.value = c.idCredito; 
                    opt.text = `${c.idCredito} - ${c.nombreSocio} (Resta: $${c.saldoPendiente.toFixed(2)})`; 
                    select.appendChild(opt); 
                });
            }).obtenerCreditosVigentes();
        }

        function procesarAbono() {
            var cr = document.getElementById('abono-credito').value;
            var m = document.getElementById('abono-monto').value; 
            var co = document.getElementById('abono-concepto').value;
            
            if(!cr) return mostrarToast("No hay ningún crédito seleccionado.");
            if(!m || parseFloat(m) <= 0) return mostrarToast("Por favor ingresa un monto válido.");
            google.script.run.withSuccessHandler(function(r){ 
                mostrarToast(r.mensaje); 
                document.getElementById('abono-monto').value = '';
                actualizarDashboard(); cargarSeccionAbonos(); 
            }).registrarAbonoPrestamo(cr, m, co);
        }
